/*Purpose of this program is to find square of X's*/
#include <stdio.h>

int main()
{
	int i,n,j;
	printf("Enter a number:");
	scanf("%d",&n);
	for(i=1,i<=n,i++)
	{
		for(j=1;j<=n;j++)
			printf("X");
		printf("\n");
	}
	return 0;
}
